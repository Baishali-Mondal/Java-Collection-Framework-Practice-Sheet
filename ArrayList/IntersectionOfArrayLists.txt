        list1.add(1);
        list1.add(2);
        list1.add(3);
        list1.add(4);
        list1.add(8);

        ArrayList<Integer> list2 = new ArrayList<>();
        list2.add(4);
        list2.add(5);
        list2.add(2);

//        if(list1.contains(list2)) //This is comparing with the entire list and not the specific element
//            if(list1.containsAll(list2))// Do as above

        list1.retainAll(list2);//Built in method of Java Collections for intersection

    }
}
